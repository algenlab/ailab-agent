Algorithm Code (Python)
def find_provinces(isConnected): n = len(isConnected) parent = list(range(n)) def find(x): if parent[x] != x: parent[x] = find(parent[x]) return parent[x] def union(x, y): root_x = find(x) root_y = find(y) if root_x != root_y: parent[root_y] = root_x for i in range(n): for j in range(i + 1, n): if isConnected[i][j] == 1: union(i, j) provinces = set() for i in range(n): provinces.add(find(i)) return len(provinces)
