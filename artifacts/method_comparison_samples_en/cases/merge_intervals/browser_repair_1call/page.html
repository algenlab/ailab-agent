function mergeIntervals(intervals) { if (intervals.length === 0) return []; intervals.sort((a, b) => a[0] - b[0]); // Sort by start time let merged = [intervals[0]]; // Initialize with first interval for (let i = 1; i < intervals.length; i++) { let curr = intervals[i]; let last = merged[merged.length - 1]; if (curr[0] <= last[1]) { // Overlap or contiguous last[1] = Math.max(last[1], curr[1]); // Merge } else { merged.push(curr); // No overlap, add new interval } } return merged; // Return final merged list }
