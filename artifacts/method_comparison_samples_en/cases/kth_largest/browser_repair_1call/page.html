Algorithm Code

def findKthLargest(nums, k): heap = [] # min-heap of capacity k for num in nums: if len(heap) < k: heapq.heappush(heap, num) # add directly elif num > heap[0]: heapq.heapreplace(heap, num) # replace top # else: num <= heap[0], skip it return heap[0] # top = k-th largest
