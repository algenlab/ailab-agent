function solveMaxIndependentSet(tree):
    root = choose_root(tree)        # node 1
    dp_take = {}, dp_skip = {}
    function dfs(u):
        dp_take[u] = value[u]
        dp_skip[u] = 0
        for each child v of u:
            dfs(v)
            dp_take[u] += dp_skip[v]
            dp_skip[u] += max(dp_take[v], dp_skip[v])
    dfs(root)
    return max(dp_take[root], dp_skip[root])
