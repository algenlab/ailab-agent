💻 算法代码（#code）

function findCutVerticesAndBridges(graph) { let dfn = {}, low = {}, parent = {}, timer = 0; let articulation = new Set(), bridges = []; function dfs(u) { dfn[u] = low[u] = ++timer; let children = 0; for (let v of graph[u]) { if (dfn[v] === undefined) { parent[v] = u; children++; dfs(v); low[u] = Math.min(low[u], low[v]); if (low[v] > dfn[u]) bridges.push([u, v]); if (low[v] >= dfn[u] && parent[u] !== undefined) articulation.add(u); } else if (v !== parent[u]) { low[u] = Math.min(low[u], dfn[v]); } } if (parent[u] === undefined && children > 1) articulation.add(u); } dfs('A'); return {articulation: [...articulation], bridges}; }
