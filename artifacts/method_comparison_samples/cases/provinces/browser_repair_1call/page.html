省份数量
LeetCode 547 · 并查集 (Union-Find)
当前省份数: 3 解法: 并查集 · O(n²)

📜 算法代码 (JavaScript)

function find(parent, i) {
                        if (parent[i] === i) return i;
                        return find(parent, parent[i]);
                    }
                    function union(parent, x, y) {
                        let rootX = find(parent, x);
                        let rootY = find(parent, y);
                        if (rootX !== rootY) {
                            parent[rootX] = rootY;
                            return true;
                        }
                        return false;
                    }
                    function solve(isConnected) {
                        const n = isConnected.length;
                        const parent = Array.from({length: n}, (_, i) => i);
                        let count = n;
                        for (let i = 0; i < n; i++) {
                            for (let j = i + 1; j < n; j++) {
                                if (isConnected[i][j] === 1) {
                                    if (union(parent, i, j)) {
                                        count--;
                                    }
                                }
                            }
                        }
                        return count;
                    }
