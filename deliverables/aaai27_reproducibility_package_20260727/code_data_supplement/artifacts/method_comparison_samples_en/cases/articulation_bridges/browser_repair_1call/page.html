function findArticulationPointsAndBridges(graph):
    dfn = {}  // discovery time
    low = {}  // lowest reachable dfn
    parent = {}  // DFS tree parent
    time = 0
    articulation = []
    bridges = []

    function dfs(u):
        dfn[u] = low[u] = ++time
        children = 0
        for each neighbor v of u:
            if v not in dfn:           // tree edge
                parent[v] = u
                children++
                dfs(v)
                low[u] = min(low[u], low[v])
                if low[v] > dfn[u]:
                    bridges.push([u, v])
                if low[v] >= dfn[u] and parent[u] is not null:
                    articulation.push(u)
            else if v != parent[u]:     // back edge
                low[u] = min(low[u], dfn[v])
        if parent[u] is null and children > 1:
            articulation.push(u)

    for each node in graph:
        if node not in dfn:
            dfs(node)
    return {articulation, bridges}
