Pseudocode

function uniquePaths(m, n):
                        dp = new 2D array[m][n]
                        // Initialize base cases
                        for i = 0 to m-1: dp[i][0] = 1
                        for j = 0 to n-1: dp[0][j] = 1
                        // Fill the DP table
                        for i = 1 to m-1:
                            for j = 1 to n-1:
                                dp[i][j] = dp[i-1][j] + dp[i][j-1]
                        return dp[m-1][n-1]
