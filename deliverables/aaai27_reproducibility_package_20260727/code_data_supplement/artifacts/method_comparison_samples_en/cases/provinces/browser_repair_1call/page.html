def numberOfProvinces(isConnected): n = len(isConnected) parent = list(range(n)) def find(x): if parent[x] != x: parent[x] = find(parent[x]) return parent[x] def union(x, y): rootX = find(x) rootY = find(y) if rootX != rootY: parent[rootX] = rootY for i in range(n): for j in range(i+1, n): if isConnected[i][j] == 1: union(i, j) provinces = set() for i in range(n): provinces.add(find(i)) return len(provinces)
